## There is / there are
$$
\begin{array}{|l|l|l|}
\hline
& \textbf{Singular} & \textbf{Plural} \\
\hline
\fbox{+} & \textbf{There's}\text{ a piano.} & \textbf{There are}\text{ some glasses in the cupboard.} \\
\hline
\fbox{-} & \textbf{There isn't}\text{ a fridge.} & \textbf{There aren't}\text{ any pictures.} \\
\hline
\fbox{?} & \textbf{Is there}\text{ a TV?} & \textbf{Are there}\text{ any glasses?} \\
\fbox{$\checkmark$}\text{ }\fbox{$\times$} & \text{Yes,}\textbf{ there is.}\text{ No,}\textbf{ there isn't.} & \text{Yes,}\textbf{ there are.}\text{ No,}\textbf{ there aren't.} \\
\hline
\end{array}
$$

## There was / there were
$$
\begin{array}{|l|l|l|}
\hline
& \textbf{Singular} & \textbf{Plural} \\
\hline
\fbox{+} & \textbf{There was}\text{ an old TV.} & \textbf{There were}\text{ only three guests.} \\
\hline
\fbox{-} & \textbf{There wasn't}\text{ a remote control.} & \textbf{There weren't}\text{ any more people.} \\
\hline
\fbox{?} & \textbf{Was there}\text{ a ghost?} & \textbf{Were there}\text{ any lights?} \\
\fbox{$\checkmark$}\text{ }\fbox{$\times$} & \text{Yes,}\textbf{ there was.}\text{ No,}\textbf{ there wasn't.} & \text{Yes,}\textbf{ there were.}\text{ No,}\textbf{ there weren't.} \\
\hline
\end{array}
$$

## Present continuous: *be + verb + ing*
### $\fbox{+}$
$$
\begin{array}{|l|l|l|}
\hline
\textbf{Full form} & \textbf{Contraction} & \\
\hline
\text{I}\textbf{ am} & \text{I}\textbf{'m} & \\
\text{You}\textbf{ are} & \text{You}\textbf{'re} & \textbf{crying.} \\
\text{He / She / It}\textbf{ is} & \text{He / She / It}\textbf{'s} & \textbf{having}\text{ a party.} \\
\text{We}\textbf{ are} & \text{We}\textbf{'re} & \textbf{arguing.} \\
\text{You}\textbf{ are} & \text{You}\textbf{'re} & \\
\text{They}\textbf{ are} & \text{They}\textbf{'re} & \\
\hline
\end{array}
$$

### $\fbox{-}$
$$
\begin{array}{|l|l|l|}
\hline
\textbf{Full form} & \textbf{Contraction} & \\
\hline
\text{I}\textbf{ am not} & \text{I}\textbf{'m not} & \\
\text{You}\textbf{ are not} & \text{You}\textbf{'re not} & \textbf{crying.} \\
\text{He / She / It}\textbf{ is not} & \text{He / She / It}\textbf{ isn't} & \textbf{having}\text{ a party.} \\
\text{We}\textbf{ are not} & \text{We}\textbf{ aren't} & \textbf{arguing.} \\
\text{You}\textbf{ are not} & \text{You}\textbf{ aren't} & \\
\text{They}\textbf{ are not} & \text{They}\textbf{ aren't} & \\
\hline
\end{array}
$$

$$
\begin{array}{|l|l|l|l|l|l|}
\hline
\fbox{?} & & & \fbox{$\checkmark$} & & \fbox{$\times$} & \\

$$
