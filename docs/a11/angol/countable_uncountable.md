# Countable / uncountable
## *a / an, some / any*
$$
\begin{array}{|l|l|l|}
\hline
& \textbf{Countable} & \textbf{Uncountable} \\
\hline
\fbox{+}\text{ We need} & \textbf{an}\text{ apple.} & \textbf{some}\text{ butter.} \\
& \textbf{some}\text{ apples.} & \textbf{some}\text{ milk.} \\
\hline
\fbox{-}\text{ We don't need} & \textbf{a}\text{ tomato.} & \textbf{any}\text{ rice.} \\
& \textbf{any}\text{ tomato}\textbf{es}\text{.} & \textbf{any}\text{ sugar.} \\
\hline
\fbox{?}\text{ Do we need} & \textbf{a}\text{ tomato?} & \textbf{any}\text{ rice?} \\
& \textbf{any}\text{ tomato}\textbf{es}\text{?} & \textbf{any}\text{ sugar?} \\
\hline
\end{array}
$$

## *how much / how many?*
$$
\begin{array}{|l|l|l|}
\hline
\textbf{Uncountable}\text{ (singular)} & \textbf{Full answers} & \textbf{Short answers} \\
\hline
\textbf{How much}\text{ water} & \text{I drink}\textbf{ a lot of}\text{ water.} & \textbf{A lot.} \\
\text{do you drink?} & \text{I drink}\textbf{ quite a lot of}\text{ water.} & \textbf{Quite a lot.} \\
& \text{I don't drink}\textbf{ much}\text{ water.} & \textbf{Not much.} \\
& \text{I don't drink}\textbf{ any}\text{ water.} & \textbf{None.} \\
\hline
\textbf{Countable}\text{ (plural)} & & \\
\hline
\textbf{How many}\text{ sweets} & \text{I ate}\textbf{ a lot of}\text{ sweets.} & \textbf{A lot.} \\
\text{did you eat?} & \text{I ate}\textbf{ quite a lot of}\text{ sweets.} & \textbf{Quite a lot.} \\
& \text{I didn't eat}\textbf{ many}\text{ sweets.} & \textbf{Not many.} \\
& \text{I didn't eat}\textbf{ any}\text{ sweets.} & \textbf{None.} \\
\hline
\end{array}
$$

## *be going to* (plans)
$\fbox{+}$

$$
\begin{array}{|l|l|l|}
\hline
\textbf{Full form} & \textbf{Contraction} & \\
\text{I}\textbf{ am} & \text{I}\textbf{'m} & \\
\text{You}\textbf{ are} & \text{You}\textbf{'re} & \\
\text{He / She / It}\textbf{ is} & \text{He / She / It}\textbf{'s} & \textbf{going to have} \\
\text{We}\textbf{ are} & \text{We}\textbf{'re} & \text{a holiday} \\
\text{You}\textbf{ are} & \text{You}\textbf{'re} & \text{next month.} \\
\text{They}\textbf{ are} & \text{They}\textbf{'re} & \\
\hline
\end{array}
$$

$\fbox{-}$
$$
\begin{array}{|l|l|l|}
\hline
\textbf{Full form} & \textbf{Contraction} & \\
\hline
\text{I}\textbf{ am not} & \text{I}\textbf{'m not} & \\
\text{You}\textbf{ are not} & \text{You}\textbf{ aren't} & \textbf{going to have} \\
\text{He / She / It}\textbf{ is not} & \text{He / She / It}\textbf{ isn't} & \text{a holiday} \\
\text{We}\textbf{ are not} & \text{We}\textbf{ aren't} & \text{next month.} \\
\text{You}\textbf{ are not} & \text{You}\textbf{ aren't} & \\
\text{They}\textbf{ are not} & \text{They}\textbf{ aren't} & \\
\hline
\end{array}
$$
