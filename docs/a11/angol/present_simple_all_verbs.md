## present simple all verbs $\fbox{+}$ and $\fbox{-}$
$$
\begin{array}{|l|l|}
\hline
\fbox{+} & \fbox{-} \\
\hline
\text{I}\textbf{ work.} & \text{I}\textbf{ don't work.} \\
\text{You}\textbf{ work.} & \text{You}\textbf{ don't work.} \\
\text{He / She / It}\textbf{ works.} & \text{He / She / It}\textbf{ doesn't work.} \\
\text{We}\textbf{ work.} & \text{We}\textbf{ don't work.} \\
\text{You}\textbf{ work.} & \text{You}\textbf{ don't work.} \\
\text{They}\textbf{ work.} & \text{They}\textbf{ don't work.} \\
\hline
\end{array}
$$

### spelling rules for *he / she / it*
$$
\begin{array}{|l|l|}
\hline
\text{I work / play / live.} & \text{He}\textbf{ works / plays / lives.} \\
\text{I watch / finish / go / do.} & \text{He}\textbf{ watches / finishes / goes / does.} \\
\text{I study.} & \text{He}\textbf{ studies.} \\
\hline
\end{array}
$$

## present simple all verbs $\fbox{?}$
$$
\begin{array}{|l|l|l|l|l|}
\hline
\fbox{?} & \fbox{$\checkmark$} & & \fbox{$\times$} & \\
\hline
\textbf{Do}\text{ I work?} & & \text{I}\textbf{ do.} & & \text{I}\textbf{ don't.} \\
\textbf{Do}\text{ you work?} & & \text{you}\textbf{ do.} & & \text{you}\textbf{ don't.} \\
\textbf{Does}\text{ he / she / it work?} & \text{Yes,} & \text{he / she / it}\textbf{ does.} & \text{No,} & \text{he / she / it}\textbf{ doesn't.} \\
\textbf{Do}\text{ we work?} & & \text{we}\textbf{ do.} & & \text{we}\textbf{ don't.} \\
\textbf{Do}\text{ you work?} & & \text{you}\textbf{ do.} & & \text{you}\textbf{ don't.} \\
\textbf{Do}\text{ they work?} & & \text{they}\textbf{ do.} & & \text{they}\textbf{ don't.} \\
\hline
\end{array}
$$

## possessive *s*
$$
\begin{array}{|l|}
\hline
\text{She is Naomi Cambell}\textbf{'s}\text{ mother.} \\
\text{It's James}\textbf{'s}\text{ house.} \\
\text{They are my parents}\textbf{s'}\text{ friends.} \\
\hline
\end{array}
$$
